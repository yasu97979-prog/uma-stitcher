import IPython
from google.colab import output
import base64
import os
import shutil

# 画像保存用フォルダの準備（リセット）
image_dir = '/content/uma_images'
if os.path.exists(image_dir):
    shutil.rmtree(image_dir)
os.makedirs(image_dir, exist_ok=True)

# 画像IDカウンター（削除しても重複しないように管理）
global image_id_counter
image_id_counter = 0

def receive_image(b64_string):
    global image_id_counter
    image_id_counter += 1
    
    header, encoded = b64_string.split(",", 1)
    data = base64.b64decode(encoded)
    
    filename = f"{image_dir}/{image_id_counter}.png"
    with open(filename, "wb") as f:
        f.write(data)
    
    # 現在保存されているファイル数を数えて返す
    count = len(os.listdir(image_dir))
    return f"{image_id_counter},{count}"

def delete_image(image_id):
    filename = f"{image_dir}/{image_id}.png"
    if os.path.exists(filename):
        os.remove(filename)
    count = len(os.listdir(image_dir))
    return str(count)

output.register_callback('notebook.receive_image', receive_image)
output.register_callback('notebook.delete_image', delete_image)
